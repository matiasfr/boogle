export {};
const { makeBot, hideBotNotes } = await import('/dev/look/bot.ts' as string);
const { me } = await import('/src/player/player.ts' as string);
const { teleport } = await import('/src/physics/engine.ts' as string);
const { inspectRailway, railReady } = await import('/dev/look/rail-read.ts' as string);
const L = await import('/src/place/layout.ts' as string);
const bot = makeBot({recover:()=>{throw new Error('forbidden walking recovery')}});
if(!await railReady()) throw new Error('not ready');
const up=inspectRailway()!.line2View.upPlatforms.find((u:any)=>u.pair===5);
if(!up?.up) throw new Error('west platform absent');
const at=(s:number,lat:number,dy:number)=>{const p=L.lineAt(L.LINKS[1].line,s,{x:0,z:0,ux:0,uz:0});return {x:p.x+L.ORIGIN.x-p.uz*lat,z:p.z+p.ux*lat,y:L.railY(L.LINKS[1].line,s)+dy}};
const start=at(720,-6.95,1.6);
teleport(me.body!,start.x,start.y,start.z);me.body!.velocity.set(0,0,0);
await bot.until(()=>me.coyote>0,20000,'landed on west platform');
const camera=at(713,-21,14),target=at(712,-1.4,.5),w=window as any;
const set=()=>w.__debugCam?.set([camera.x,camera.y,camera.z],[target.x,target.y,target.z],68);
setInterval(set,30);set();
w.__pierWalk=async()=>{
 const path=[...up.path,at(713,1.5,0),at(713,2.9,0),at(719,2.9,0)];
 const walk=async(p:any,label:string)=>{if(!await bot.walk(p.x,p.z,.45))throw new Error(label);if(me.swimming)throw new Error('swam '+label);console.log('PLAY PIER_WALK '+JSON.stringify({label,...me.body!.position}));};
 console.log('PLAY PIER_WALK_START '+JSON.stringify(me.body!.position));
 for(const [i,p] of path.entries())await walk(p,'out '+i);
 await bot.wait(700);
 for(const [i,p] of [...path].reverse().entries())await walk(p,'back '+i);
 await walk(start,'west lever');
 console.log('PLAY PIER_WALK_PASS both directions, ramp, crossing, east ramp, west lever; no recovery');
 w.__pierWalkDone=true;
};
hideBotNotes();
w.__pierWalkReady=true;
