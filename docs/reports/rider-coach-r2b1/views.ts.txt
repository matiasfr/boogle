export {};
const { makeBot, hideBotNotes } = await import('/dev/look/bot.ts' as string);
const { me } = await import('/src/player/player.ts' as string);
const { teleport } = await import('/src/physics/engine.ts' as string);
const { interactTarget } = await import('/src/player/hands.ts' as string);
const L = await import('/src/place/layout.ts' as string);
const { inspectRailway, railReady } = await import('/dev/look/rail-read.ts' as string);
const bot = makeBot();
if (!(await railReady())) throw new Error('rail not ready');
const q = new URLSearchParams(location.search), shot = q.get('pierShot') ?? 'top';
const at = (s: number, lat: number, up: number) => {
  const p = L.lineAt(L.LINKS[1].line, s, { x: 0, z: 0, ux: 0, uz: 0 });
  return [p.x + L.ORIGIN.x - p.uz * lat, L.railY(L.LINKS[1].line, s) + up, p.z + p.ux * lat];
};
const pos = at(714, 1.4, 1);
teleport(me.body!, ...pos); me.body!.velocity.set(0, 0, 0);
const cameras = {
  lever: [at(720.5,-7.3,2.42),at(706,-4,0.1),70],
  east: [at(706,2,1.7),at(710,-5.5,0.45),75],
  top: [at(719,-2,43),at(719,-2,0),65],
  south: [at(690,1.3,1.7),at(717,-2,1.1),68],
  car: [[1398.7506398967087,26.55,-354.7871484643708],at(709,-6,0.4),75],
  long: [at(707.8,4,2.3),at(716,-4,0.3),75],
};
const [camera,target,fov] = cameras[shot];
const w = window as any;
const set = () => w.__debugCam?.set(camera,target,fov);
setInterval(set,30);set();
if (shot === 'car' || shot === 'long') {
  const view = () => inspectRailway()?.line2View;
  const halt = () => view()?.halts?.find((h:any)=> h.name === 'Pier');
  const knob = () => halt()?.lever.find((k:any)=>k.platform === 'up' && k.on);
  if (!(await bot.until(knob,60000,'Pier lever'))) throw new Error('no lever');
  const k=knob(), p=at(720,-7.1,1.3); teleport(me.body!,...p); me.body!.velocity.set(0,0,0);
  const anyStanding=()=>view()?.trains.find((t:any)=>t.line===1&&t.track===1&&t.phase===0&&t.at===3);
  const desired=()=>{const t=anyStanding();return t&&(shot!=='long'||(t.wagon?.shown&&t.longBoiler))?t:null};
  await bot.wait(1200);
  for(let n=0;n<16&&!desired();n++) {
    if(anyStanding()) {
      if(!await bot.until(()=>!anyStanding(),2400000,'other train leaves'))throw new Error('other train stayed');
    }
    if(!knob()?.over) {
      if(!await bot.until(()=>interactTarget()?.run,2400000,'lever ready'))throw new Error('lever blocked');
      if(!await bot.use(k.x,k.z,'request homeward',/^Request stop: to /))throw new Error('lever unavailable');
    }
    if(!await bot.until(()=>anyStanding(),2400000,'homeward train stands'))throw new Error('no train');
    console.log('PLAY PIER_REQUEST '+JSON.stringify({n,slot:anyStanding()?.slot,wagon:!!anyStanding()?.wagon?.shown,long:anyStanding()?.longBoiler}));
  }
  if(!desired())throw new Error('long wagon train never stood');
  console.log('PLAY PIER_STANDING '+JSON.stringify(desired()));
}
console.log('PLAY PIER_CAMERA '+JSON.stringify({shot,camera,target,fov}));
hideBotNotes();
w.__pierShotReady = true;
