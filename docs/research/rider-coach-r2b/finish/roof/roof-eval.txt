function flight(rise,press){let y=0,v=JUMP_SPEED,pressed=false,peak=0;for(let t=0;t<5;t+=dt){if(!pressed&&t>=press){v=Math.max(v,air);pressed=true;}const g=v<0?JUMP.fallGravity:v<(pressed?apex:JUMP.apexSpeed)?APEX_GRAVITY:1;v-=GRAVITY*g*dt;y+=v*dt;peak=Math.max(peak,y);if(pressed&&v<0&&y<rise)return peak>=rise?t:0;}return 0;}
// Sample the air-jump press every 1/60 s, integrating the production gravity/apex rules at 240 Hz.
const cache=new Map();
function reach(rise){const key=Math.round(rise*20);if(cache.has(key))return cache.get(key);let t=0;for(let p=0;p<1.5;p+=1/60)t=Math.max(t,flight(key/20,p));const d=t*9.5;cache.set(key,d);return d;}
